```ts
// stores/openSourceStore.ts
import { create } from 'zustand'

interface OpenSourceState {
  // 搜索页筛选快照（用于从概览页跳转时恢复）
  lastSearchQuery: OSSearchQuery | null
  setLastSearchQuery: (query: OSSearchQuery | null) => void

  // 收藏 IDs 本地缓存（与学术子系统隔离）
  favoriteDeveloperIds: Set<number>
  setFavoriteDeveloperIds: (ids: number[]) => void
  toggleFavoriteDeveloper: (id: number) => void
}

export const useOpenSourceStore = create<OpenSourceState>((set) => ({
  lastSearchQuery: null,
  setLastSearchQuery: (query) => set({ lastSearchQuery: query }),

  favoriteDeveloperIds: new Set(),
  setFavoriteDeveloperIds: (ids) =>
    set({ favoriteDeveloperIds: new Set(ids) }),
  toggleFavoriteDeveloper: (id) =>
    set((state) => {
      const next = new Set(state.favoriteDeveloperIds)
      if (next.has(id)) next.delete(id)
      else next.add(id)
      return { favoriteDeveloperIds: next }
    }),
}))
```
