## 📋 文档说明

### 适用场景
本设计适用于**内部团队使用**的智能人才库系统，主要特点：
- 用户范围可控（已知的招聘团队成员）
- 数据主要为公开学术信息
- 无金融交易等高风险操作
- 但仍有质量和数据安全要求

### 设计原则（补充）
1. **实用优先**：在保证基本安全的前提下，简化不必要的复杂性
2. **渐进增强**：先实现核心安全控制，再根据实际需要增强
3. **透明可查**：安全规则明确，操作可追溯
4. **风险适配**：安全措施与内部使用风险等级相匹配

## 👥 3. 角色模型

### 3.1 角色定义
根据内部使用场景，定义以下三种核心角色：

| 角色 | 英文标识 | 主要职责 | 用户示例 |
|------|----------|----------|----------|
| **普通用户** | `user` | 查看授权范围内的人才信息，进行搜索、筛选、导出等操作 | 招聘专员、招聘经理 |
| **管理员** | `admin` | 管理系统用户、配置权限、查看系统日志 | 系统负责人、HR技术管理员 |
| **超级管理员** | `super_admin` | 系统最高权限，可进行所有配置和查看所有数据 | 项目负责人、技术负责人 |

### 3.2 角色权限概览
```
普通用户 (user)
├── 查看授权学校的人才数据
├── 搜索、筛选、排序人才
├── 查看人才详情
├── 收藏关注的人才
└── 导出授权范围内的数据

管理员 (admin)
├── 具备所有普通用户权限
├── 管理用户账号（增删改查）
├── 配置用户学校权限
├── 查看系统操作日志
└── 查看数据同步状态

超级管理员 (super_admin)
├── 具备所有管理员权限
├── 管理所有管理员账号
├── 查看所有数据（无视权限过滤）
├── 系统配置管理
└── 数据库维护操作
```

### 3.3 角色分配原则
- **默认角色**：新用户默认为`user`角色
- **晋升流程**：`user` → `admin` 需经现有管理员审批
- **最小权限**：只授予完成工作所需的最小权限
- **定期审查**：每季度审查角色分配是否合理

def check_user_permission(user_id, school_id):
    """
    检查用户是否有权限访问指定学校
    """
    # 1. 获取用户所有有效授权
    scopes = get_user_scopes(user_id)

    # 2. 按优先级检查
    for scope in scopes:
        if scope['scope_type'] == 'all':
            return True  # 有全局权限

        if scope['scope_type'] == 'country':
            school_country = get_school_country(school_id)
            if school_country == scope['scope_value']:
                return True  # 有该国权限

        if scope['scope_type'] == 'school':
            if school_id == scope['scope_value']:
                return True  # 有该校权限

    return False  # 无权限

def get_user_accessible_schools(user_id):
    """
    获取用户可访问的所有学校列表
    """
    scopes = get_user_scopes(user_id)
    accessible_schools = set()

    for scope in scopes:
        if scope['scope_type'] == 'all':
            # 所有学校
            return get_all_schools()

        elif scope['scope_type'] == 'country':
            # 该国的所有学校
            country_schools = get_schools_by_country(scope['scope_value'])
            accessible_schools.update(country_schools)

        elif scope['scope_type'] == 'school':
            accessible_schools.add(scope['scope_value'])

    return list(accessible_schools)
```

# FastAPI权限装饰器示例
from functools import wraps
from fastapi import HTTPException, Depends
from fastapi.security import HTTPBearer, HTTPAuthorizationCredentials

security = HTTPBearer()

def require_role(required_role):
    """
    角色权限检查装饰器
    """
    def decorator(func):
        @wraps(func)
        async def wrapper(
            *args,
            credentials: HTTPAuthorizationCredentials = Depends(security),
            **kwargs
        ):
            # 验证Token并获取用户信息
            user = authenticate_user(credentials.credentials)

            # 检查角色
            if user["role"] not in required_role:
                raise HTTPException(
                    status_code=403,
                    detail=f"需要{required_role}角色，当前角色：{user['role']}"
                )

            return await func(*args, user=user, **kwargs)
        return wrapper
    return decorator

def require_school_permission(school_id_param="school_id"):
    """
    学校权限检查装饰器
    """
    def decorator(func):
        @wraps(func)
        async def wrapper(
            *args,
            user: dict = Depends(get_current_user),
            **kwargs
        ):
            # 从参数中获取school_id
            school_id = kwargs.get(school_id_param)

            if not school_id:
                raise HTTPException(status_code=400, detail="缺少school_id参数")

            # 检查权限
            if not check_user_permission(user["user_id"], school_id):
                raise HTTPException(
                    status_code=403,
                    detail=f"无权限访问学校{school_id}"
                )

            return await func(*args, user=user, **kwargs)
        return wrapper
    return decorator

# 敏感信息脱敏处理
def mask_sensitive_data(data):
    """
    脱敏敏感信息
    """
    if "email" in data:
        # 邮箱脱敏：ab***@example.com
        email = data["email"]
        if "@" in email:
            username, domain = email.split("@", 1)
            if len(username) > 2:
                masked = username[:2] + "*" * (len(username) - 2)
            else:
                masked = "*" * len(username)
            data["email"] = f"{masked}@{domain}"

    if "phone" in data:
        # 手机号脱敏：138****5678
        phone = data["phone"]
        if len(phone) >= 11:
            data["phone"] = phone[:3] + "*" * 4 + phone[-4:]

    return data

# 日志脱敏
class SensitiveDataFilter(logging.Filter):
    def filter(self, record):
        # 脱敏日志中的敏感信息
        if hasattr(record, 'msg'):
            record.msg = self.mask_sensitive_info(record.msg)
        return True

    def mask_sensitive_info(self, text):
        patterns = [
            (r'password["\']?\s*[:=]\s*["\']?([^"\'\s]+)', 'password="***"'),
            (r'token["\']?\s*[:=]\s*["\']?([^"\'\s]+)', 'token="***"'),
            (r'api_key["\']?\s*[:=]\s*["\']?([^"\'\s]+)', 'api_key="***"'),
        ]

        for pattern, replacement in patterns:
            text = re.sub(pattern, replacement, text, flags=re.IGNORECASE)

        return text
```

def validate_input(schema):
    """
    输入验证装饰器
    """
    def decorator(func):
        @wraps(func)
        async def wrapper(*args, **kwargs):
            # 获取请求数据
            request_data = kwargs.get("data") or {}

            # 验证数据
            try:
                validated_data = schema.validate(request_data)
                kwargs["data"] = validated_data
            except ValidationError as e:
                raise HTTPException(status_code=400, detail=str(e))

            return await func(*args, **kwargs)
        return wrapper
    return decorator

### 7.2 审计查询与分析

#### 7.2.1 审计查询接口
```python
@app.get("/audit/logs")
@require_role(["admin", "super_admin"])
async def get_audit_logs(
    start_time: Optional[datetime] = None,
    end_time: Optional[datetime] = None,
    user_id: Optional[str] = None,
    event_type: Optional[str] = None,
    resource_type: Optional[str] = None,
    page: int = 1,
    page_size: int = 50
):
    """
    查询审计日志（管理员权限）
    """
    # 构建查询条件
    query_filters = []

    if start_time:
        query_filters.append(AuditLog.event_time >= start_time)
    if end_time:
        query_filters.append(AuditLog.event_time <= end_time)
    if user_id:
        query_filters.append(AuditLog.user_id == user_id)
    if event_type:
        query_filters.append(AuditLog.event_type == event_type)
    if resource_type:
        query_filters.append(AuditLog.resource_type == resource_type)

    # 执行查询（分页）
    query = AuditLog.query.filter(*query_filters)
    total = query.count()
    logs = query.order_by(AuditLog.event_time.desc()) \
               .offset((page - 1) * page_size) \
               .limit(page_size) \
               .all()

    return {
        "total": total,
        "page": page,
        "page_size": page_size,
        "data": [log.to_dict() for log in logs]
    }
```
