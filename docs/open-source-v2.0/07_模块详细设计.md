### 2.3 Token 轮换算法

```python
class TokenRotator:
    """
    线程安全�?Token 轮换器�?    每个 Token 独立跟踪剩余请求配额�?    """
    def __init__(self, tokens: list[str]):
        self._tokens = tokens
        self._quota: dict[str, int] = {t: 5000 for t in tokens}  # 默认 5000
        self._lock = asyncio.Lock()
        self._current_idx = 0

    async def get_token(self) -> str:
        async with self._lock:
            # 寻找剩余配额最多的 Token
            available = [(t, q) for t, q in self._quota.items() if q > 10]
            if not available:
                raise GitHubRateLimitError("All tokens exhausted")
            token, _ = max(available, key=lambda x: x[1])
            self._quota[token] -= 1
            return token

    async def update_quota_from_headers(self, token: str, headers: dict) -> None:
        """从响应头解析剩余配额"""
        remaining = headers.get("X-RateLimit-Remaining")
        if remaining is not None:
            async with self._lock:
                self._quota[token] = int(remaining)
```

# services/search/keyword.py
async def keyword_search(
    session: AsyncSession,
    query: str,
    filters: OSSearchFilters | None,
    sort_by: str,
    page: int,
    page_size: int,
) -> PaginatedResponse[OSDeveloper]:
    """
    使用 ILIKE �?name / bio / company / location / github_login 上匹配�?    """
    conditions = []
    if query:
        q = f"%{query}%"
        conditions.append(
            or_(
                OSDeveloper.name.ilike(q),
                OSDeveloper.bio.ilike(q),
                OSDeveloper.company.ilike(q),
                OSDeveloper.location.ilike(q),
                OSDeveloper.github_login.ilike(q),
            )
        )

    if filters:
        if filters.tech_elements:
            conditions.append(
                OSDeveloper.tech_tags.overlap(filters.tech_elements)
            )
        if filters.languages:
            conditions.append(
                OSDeveloper.primary_languages.overlap(filters.languages)
            )
        if filters.location:
            conditions.append(OSDeveloper.location.ilike(f"%{filters.location}%"))
        if filters.company:
            conditions.append(OSDeveloper.company.ilike(f"%{filters.company}%"))
        if filters.min_stars is not None:
            conditions.append(OSDeveloper.total_stars_received >= filters.min_stars)

    stmt = select(OSDeveloper).where(and_(*conditions))

    # 排序
    order_map = {
        "stars_desc": OSDeveloper.total_stars_received.desc(),
        "stars_asc": OSDeveloper.total_stars_received.asc(),
        "name_asc": OSDeveloper.name.asc(),
    }
    stmt = stmt.order_by(order_map.get(sort_by, OSDeveloper.total_stars_received.desc()))

    # 分页
    total = await session.scalar(select(func.count()).select_from(stmt.subquery()))
    stmt = stmt.offset((page - 1) * page_size).limit(page_size)

    result = await session.execute(stmt)
    items = result.scalars().all()

    return PaginatedResponse(
        data=items,
        total=total or 0,
        page=page,
        page_size=page_size,
    )

# services/search/hybrid.py
async def hybrid_search(
    session: AsyncSession,
    query: str,
    filters: OSSearchFilters | None,
    embedding_service: OSEmbeddingService,
    page: int,
    page_size: int,
    k: int = 60,  # RRF 常数
) -> PaginatedResponse[OSDeveloper]:
    """
    1. 分别执行 keyword_search �?semantic_search（各�?Top N�?    2. 使用 RRF（Reciprocal Rank Fusion）融合排�?    3. 返回融合后的分页结果
    """
    top_n = page * page_size  # 取足够多以便融合后分�?
    # 并行执行两种搜索
    keyword_results, semantic_results = await asyncio.gather(
        keyword_search(session, query, filters, "stars_desc", 1, top_n),
        semantic_search(session, query, filters, embedding_service, 1, top_n),
    )

    # RRF 打分
    scores: dict[int, float] = defaultdict(float)

    for rank, dev in enumerate(keyword_results.data):
        scores[dev.developer_id] += 1.0 / (k + rank + 1)

    for rank, dev in enumerate(semantic_results.data):
        scores[dev.developer_id] += 1.0 / (k + rank + 1)

    # 按分数排�?    all_devs = {dev.developer_id: dev for dev in keyword_results.data + semantic_results.data}
    sorted_ids = sorted(scores.keys(), key=lambda id: scores[id], reverse=True)

    # 分页
    start = (page - 1) * page_size
    end = start + page_size
    page_ids = sorted_ids[start:end]
    page_items = [all_devs[id] for id in page_ids if id in all_devs]

    return PaginatedResponse(
        data=page_items,
        total=len(sorted_ids),
        page=page,
        page_size=page_size,
    )

### 6.5 取消机制

```python
class OSCollectOrchestrator:
    def __init__(self, ...):
        self._running_tasks: dict[int, asyncio.Task] = {}
        self._cancel_flags: dict[int, bool] = {}

    async def run(self, task_id: int, config: OSCollectConfig) -> None:
        self._cancel_flags[task_id] = False
        context = CollectContext(task_id=task_id, config=config)

        task = asyncio.create_task(self._run_phases(context))
        self._running_tasks[task_id] = task

        try:
            await task
        except TaskCancelledError:
            await self._update_task_status(task_id, "cancelled")

            self._cancel_flags.pop(task_id, None)

        if task := self._running_tasks.get(task_id):

    async def _run_phases(self, ctx: CollectContext) -> None:

            await self._update_task_step(ctx.task_id, f"Executing {name}")

            try:
                await phase.execute(ctx)

                if name in ("fetch_repos", "sync_serving"):
                    raise  # 关键阶段失败则终�?                # 非关键阶段记录错误继�?
            progress += weight
            await self._update_task_progress(ctx.task_id, progress)

        await self._update_task_status(ctx.task_id, "completed")
```
